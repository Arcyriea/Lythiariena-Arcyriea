## Phase 3B.3 Stage A — Folder-derived packaging

The one-shot packager derives category membership from immediate subdirectories of `private/assets/` and recursively includes registered files within each category. The current outputs are `ships.bin`, `components.bin`, `turrets-platforms.bin`, and `planets.bin`; nested source folders do not create additional bundles. The registry owns asset identity and runtime metadata, while folder names alone determine bundle membership.

Registry schema 1.1 and generated manifest schema 1.1 are independent from the unchanged RTAB 1.0 binary format. The registry contains the 31 existing SVG assets and only three approved planet PNGs (`Map-111.png`, `Map-122.png`, and `Map-692.png`); `Map-111-2.png` and `Map-122-2.png` remain public, unregistered, and outside private bundles. The generated Phaser SVG manifest remains SVG-only with 31 entries, and runtime texture keys and loading behavior are unchanged.

PNG validation accepts only non-interlaced RGBA8 images using compression and filter method 0, with dimensions up to 4096×4096 and a maximum compressed file size of 8 MiB (32 MiB aggregate). It independently limits decoded pixel bytes to 64 MiB per image and 256 MiB aggregate, and inflated scanline bytes to 64 MiB + 4096 bytes per image and 257 MiB aggregate. The validator checks chunk ordering and CRCs, requires one fully consumed zlib stream across consecutive IDAT chunks, bounds streaming decompression to the exact IHDR-derived scanline length, and validates every scanline filter byte. PNG payload bytes are preserved; the packager does not re-encode images.

Preparation uses an exclusive process lock and an append-only JSONL transaction journal. Each checksummed record carries the transaction ID and next sequence number; every output filesystem operation is preceded by a flushed intent and followed by a flushed completion. A committed-state record explicitly marks the new manifest as committed. Recovery runs only after exclusive lock acquisition, validates the entire journal and actual file hashes, then completes a validated commit or rolls back verified prior outputs. Malformed, truncated, or ambiguous journals and unexpected artifacts remain untouched for manual inspection.

An existing lock is never removed automatically, including when its PID appears inactive. To clear a stale lock, stop all asset preparation processes, inspect the lock JSON, and independently verify its owner and process-start identity. On Windows, inspect `Get-Process -Id <pid> -ErrorAction SilentlyContinue | Select-Object Id,StartTime,Path` and `Get-CimInstance Win32_Process -Filter "ProcessId = <pid>" | Select-Object ProcessId,CommandLine,CreationDate`; compare the executable, command line, and creation time with the lock metadata and confirm no preparation process is active. On Linux, compare `processStartIdentity` with `/proc/<pid>/stat` start ticks. Hostname, PID, or age alone is not proof of staleness. Only after an operator confirms the owner is no longer active, remove that exact file from the repository root with `Remove-Item -LiteralPath .\\private\\.asset-package.lock` (or the platform's equivalent); never remove journal or bundle artifacts as part of lock cleanup. If the owner cannot be verified or the identity is ambiguous, preserve the lock and escalate rather than retrying preparation. On clean clones without `private/`, validation checks for lock, journal, and orphan transaction files without recovering or modifying them.

Stage A does not add a filesystem listener, runtime hot reload, mod discovery/loading, Phaser bundle loading, or Phase 3C behavior.

function fail(message) {
    throw new Error(`Invalid asset registry: ${message}`);

function isSafeLogicalPath(logicalPath, contentType) {
    const extension = contentType === 'SVG' ? '.svg' : contentType === 'PNG' ? '.png' : null;
    return typeof logicalPath === 'string'
        && extension !== null
        && logicalPath.startsWith('assets/')
        && logicalPath.endsWith(extension)
        && !logicalPath.includes('\\')
        && logicalPath.split('/').every((segment) => segment && segment !== '.' && segment !== '..');
}

export function validateAssetRegistry(registry) {
    if (!registry || registry.registrySchemaVersion !== '1.1') {
        fail('registrySchemaVersion must be "1.1"');
    }
    if (!Array.isArray(registry.assets) || registry.assets.length === 0) {
        fail('assets must be a non-empty array');
    }
    if ('bundles' in registry) {
        fail('bundle membership is derived from source folders and must not be declared in the registry');
    }

    const ids = new Set();
    const paths = new Set();
    const keys = new Map();
    const aliases = new Map();

    for (const asset of registry.assets) {
        if (!asset || Object.keys(asset).some((key) => (
            !['assetId', 'logicalPath', 'textureKey', 'aliases', 'contentType'].includes(key)
        ))) {
            fail('assets must contain only asset identity and runtime metadata fields');
        }
        if (!asset || typeof asset.assetId !== 'string' || !/^[a-zA-Z0-9][a-zA-Z0-9._-]*$/.test(asset.assetId)) {
            fail('every asset must have a non-empty namespaced assetId using letters, digits, dot, underscore, or hyphen');
        }
        if (ids.has(asset.assetId)) {
            fail(`duplicate asset ID "${asset.assetId}"`);
        }
        ids.add(asset.assetId);

        if (asset.contentType !== 'SVG' && asset.contentType !== 'PNG') {
            fail(`unsupported content type "${asset.contentType}" for asset "${asset.assetId}"`);
        }
        if (!isSafeLogicalPath(asset.logicalPath, asset.contentType)) {
            fail(`asset "${asset.assetId}" has an invalid logicalPath`);
        }
        if (paths.has(asset.logicalPath.toLowerCase())) {
            fail(`duplicate logical path "${asset.logicalPath}"`);
        }
        paths.add(asset.logicalPath.toLowerCase());

        if (typeof asset.textureKey !== 'string' || asset.textureKey.length === 0) {
            fail(`asset "${asset.assetId}" must have an explicit textureKey`);
        }
        const normalizedKey = asset.textureKey.toLowerCase();
        if (keys.has(normalizedKey)) {
            fail(`duplicate texture key "${asset.textureKey}"`);
        }
        keys.set(normalizedKey, asset.textureKey);

        if (!Array.isArray(asset.aliases) || asset.aliases.some((alias) => typeof alias !== 'string' || !alias)) {
            fail(`asset "${asset.assetId}" aliases must be an array of non-empty strings`);
        }
        for (const alias of asset.aliases) {
            const normalizedAlias = alias.toLowerCase();
            if (keys.has(normalizedAlias)) {
                fail(`alias "${alias}" conflicts with texture key "${keys.get(normalizedAlias)}"`);
            }
            if (aliases.has(normalizedAlias)) {
                fail(`conflicting alias "${alias}" maps to multiple assets`);
            }
            aliases.set(normalizedAlias, asset.assetId);
        }

    }

    for (const [aliasPath, alias] of REQUIRED_ALIASES) {
        const asset = registry.assets.find((entry) => entry.logicalPath === aliasPath);
        if (!asset || asset.aliases.length !== 1 || asset.aliases[0] !== alias) {
            fail(`required alias mapping "${aliasPath}" -> "${alias}" is missing or changed`);
        }
    }
    if (aliases.size !== REQUIRED_ALIASES.size) {
        fail(`expected exactly ${REQUIRED_ALIASES.size} legacy aliases, found ${aliases.size}`);
    }
    for (const [alias, ownerId] of aliases) {
        if (keys.has(alias)) {
            fail(`alias "${alias}" conflicts with texture key "${keys.get(alias)}"`);
        }
        if (!ownerId) {
            fail(`alias "${alias}" has no asset owner`);
        }
    }

    return registry;
}

export async function readAssetRegistry(projectRoot = PROJECT_ROOT) {
    const registryPath = path.join(projectRoot, 'scripts', 'asset-registry.json');
    let content;
    try {
        content = await readFile(registryPath, 'utf8');
    } catch (error) {
        throw new Error(`Could not read asset registry at ${registryPath}: ${error.message}`);
    }

    let registry;
    try {
        registry = JSON.parse(content);
    } catch (error) {
        throw new Error(`Could not parse asset registry at ${registryPath}: ${error.message}`);
    }

    return validateAssetRegistry(registry);
}

export function resolveAssetSource(logicalPath, projectRoot = PROJECT_ROOT) {
    const root = resolvePhysicalDir(logicalPath);
    const contentType = /\.svg$/i.test(logicalPath) ? 'SVG' : /\.png$/i.test(logicalPath) ? 'PNG' : null;
    if (!root || !contentType || !isSafeLogicalPath(logicalPath, contentType)) {
        throw new Error(`Could not resolve canonical source for logical asset "${logicalPath}"`);
    }

    const rootRelative = path.relative(PROJECT_ROOT, root.physical);
    const sourceRoot = path.resolve(projectRoot, rootRelative);
    const suffix = root.logical
        ? logicalPath.slice(root.logical.length).replace(/^[/\\]/, '')
        : logicalPath;
    const sourcePath = path.resolve(sourceRoot, ...suffix.split('/'));
    if (!sourcePath.startsWith(`${sourceRoot}${path.sep}`)) {
        throw new Error(`Logical asset path escapes its canonical source root: "${logicalPath}"`);
    }
    return sourcePath;
}

    async function walk(directory, relativeDirectory = '') {

function resolveTextureKeys(relativePaths) {
    const groups = new Map();
    for (const relativePath of relativePaths) {
        const key = normalizeTextureKey(relativePath);
        const normalizedKey = key.toLowerCase();
        const group = groups.get(normalizedKey) ?? [];
        group.push({ relativePath, key });
        groups.set(normalizedKey, group);
    }

    const resolved = new Map();
    for (const group of groups.values()) {
        if (group.length > 1) {
            console.warn(
                `Normalized public SVG texture-key collision for: ${group.map(({ relativePath }) => relativePath).join(', ')}`
            );
        }

        for (const { relativePath, key } of group) {
            const suffix = group.length > 1
                ? `-${createHash('sha256').update(relativePath).digest('hex').slice(0, 10)}`
                : '';
            resolved.set(relativePath, `${key}${suffix}`);
        }
    }

    const usedKeys = new Map();
    for (const [relativePath, key] of resolved) {
        const normalizedKey = key.toLowerCase();
        const conflictingPath = usedKeys.get(normalizedKey);
        if (conflictingPath) {
            throw new Error(
                `Unable to safely resolve public SVG texture-key collision: ${conflictingPath} and ${relativePath} both map to "${key}"`
            );
        }
        usedKeys.set(normalizedKey, relativePath);
    }

    return resolved;
}

async function replaceMapping(mappingPath, expected, replacement) {
    const current = await readFile(mappingPath);
    if (!current.equals(Buffer.from(expected))) {
        fail('asset-roots.mjs changed after preflight; refusing to overwrite it');
    }

    const token = `${process.pid}-${randomUUID()}`;
    const temporary = `${mappingPath}.${token}.tmp`;
    const backup = `${mappingPath}.${token}.bak`;
    await writeFile(temporary, replacement, { flag: 'wx' });
    try {
        const stillCurrent = await readFile(mappingPath);
        if (!stillCurrent.equals(Buffer.from(expected))) {
            fail('asset-roots.mjs changed during migration; refusing to overwrite it');
        }
        await rename(mappingPath, backup);
        if (!(await readFile(backup)).equals(Buffer.from(expected))) {
            await rename(backup, mappingPath);
            fail('asset-roots.mjs changed while creating its rollback copy');
        }
        await rename(temporary, mappingPath);
        if (!(await readFile(mappingPath)).equals(Buffer.from(replacement))) {
            fail('updated asset-roots.mjs did not match the verified migration mapping');
        }
        return { backup, replacement: Buffer.from(replacement) };
    } catch (error) {
        await rm(temporary, { force: true });
        const sourceExists = await lstat(mappingPath).then(() => true, () => false);
        const backupExists = await lstat(backup).then(() => true, () => false);
        if (!sourceExists && backupExists) {
            await rename(backup, mappingPath);
        }
        throw error;
    }
}

function createBundle(bundleId, bundleFile, assets) {

    }

        left.assetId < right.assetId ? -1 : left.assetId > right.assetId ? 1 : 0
    ));
    const ids = new Set();

        if (ids.has(assetId)) {
            throw new Error(`Fatal: duplicate asset ID "${assetId}" detected. Asset IDs must be unique.`);
        }
        ids.add(assetId);
        return encodeAssetId(assetId);
    });

    ensureUint32(BigInt(orderedAssets.length), 'Entry count');

    const indexSize = BigInt(INDEX_HEADER_SIZE) + recordTableSize + stringTableSize + BigInt(DIGEST_SIZE);
    const payloadOffset = BigInt(HEADER_SIZE) + indexSize;

    ensureUint64(indexSize, 'Index size');
    ensureUint64(payloadOffset, 'Payload offset');
    ensureUint64(payloadOffset + payloadSize, 'Final bundle size');

    const indexBuffer = Buffer.alloc(ensureSafeNumber(indexSize, 'Index size'));
    indexBuffer.write('INDX', 0, 'ascii');
    indexBuffer.writeUInt32LE(orderedAssets.length, 4);
    indexBuffer.writeUInt16LE(RECORD_SIZE, 8);
    indexBuffer.writeUInt16LE(0, 10);
    indexBuffer.writeBigUInt64LE(stringTableOffset, 12);
    indexBuffer.writeBigUInt64LE(stringTableSize, 20);

    let nextStringOffset = 0n;
    let nextPayloadOffset = 0n;
    let stringCursor = ensureSafeNumber(stringTableOffset, 'String table offset');
    let payloadCursor = 0;

    orderedAssets.forEach((asset, index) => {
        const recordStart = INDEX_HEADER_SIZE + index * RECORD_SIZE;
        const id = idBytes[index];

        ensureUint64(nextPayloadOffset, 'Payload-relative offset');
        ensureUint64(payloadSize, 'Stored size');
        ensureUint64(payloadSize, 'Original size');

        indexBuffer.writeUInt32LE(Number(nextStringOffset), recordStart);
        indexBuffer.writeUInt32LE(id.length, recordStart + 4);
        indexBuffer.writeBigUInt64LE(nextPayloadOffset, recordStart + 8);
        indexBuffer.writeBigUInt64LE(payloadSize, recordStart + 16);
        indexBuffer.writeBigUInt64LE(payloadSize, recordStart + 24);
        indexBuffer.writeUInt16LE(CONTENT_TYPE_CODES[asset.contentType], recordStart + 32);
        indexBuffer.writeUInt16LE(0, recordStart + 34);
        indexBuffer.writeUInt32LE(0, recordStart + 36);

        id.copy(indexBuffer, stringCursor);
        stringCursor += id.length;
        payloadCursor += asset.bytes.length;
        nextStringOffset += BigInt(id.length);
        nextPayloadOffset += payloadSize;
    });

    const indexDigestOffset = indexBuffer.length - DIGEST_SIZE;
    sha256(indexBuffer.subarray(0, indexDigestOffset)).copy(indexBuffer, indexDigestOffset);

    const output = Buffer.alloc(ensureSafeNumber(payloadOffset + payloadSize, 'Bundle size'));
    output.write('RTAB', 0, 'ascii');
    output.writeUInt16LE(FORMAT_VERSION, 4);
    output.writeUInt16LE(HEADER_FLAGS, 6);
    output.writeBigUInt64LE(payloadOffset, 8);
    output.writeBigUInt64LE(BigInt(HEADER_SIZE), 16);
    output.writeBigUInt64LE(indexSize, 24);
    indexBuffer.copy(output, HEADER_SIZE);
    let outputPayloadCursor = ensureSafeNumber(payloadOffset, 'Payload offset');

        asset.bytes.copy(output, outputPayloadCursor);
        outputPayloadCursor += asset.bytes.length;
    });

export async function validateRtabBundle(bytes, bundle, expectedAssets) {
    if (!Buffer.isBuffer(bytes) || bytes.length < HEADER_SIZE) {
        throw new Error(`RTAB bundle "${bundle.bundleId}" is shorter than its ${HEADER_SIZE}-byte header`);
    }
    if (bytes.toString('ascii', 0, 4) !== 'RTAB') {
        throw new Error(`RTAB bundle "${bundle.bundleId}" has invalid magic`);
    }
    const version = bytes.readUInt16LE(4);
    if (version !== FORMAT_VERSION) {
        throw new Error(`Unsupported RTAB version 0x${version.toString(16).padStart(4, '0')}. Supported version: 1.0 (0x0100).`);
    }
    if (bytes.readUInt16LE(6) !== HEADER_FLAGS) {
        throw new Error('Unsupported RTAB header flags');
    }
    if (!bytes.subarray(32, HEADER_SIZE).every((byte) => byte === 0)) {
        throw new Error('RTAB reserved header bytes must be zero');
    }

    const payloadOffsetBig = readUInt64(bytes, 8, 'Payload offset');
    const indexOffsetBig = readUInt64(bytes, 16, 'Index offset');
    const indexSizeBig = readUInt64(bytes, 24, 'Index size');
    if (indexOffsetBig !== BigInt(HEADER_SIZE)) {
        throw new Error('RTAB index offset must be exactly 64');
    }
    if (indexSizeBig < BigInt(INDEX_HEADER_SIZE + RECORD_SIZE + DIGEST_SIZE)) {
        throw new Error('RTAB index is too small to contain an asset record');
    }
    if (indexOffsetBig + indexSizeBig !== payloadOffsetBig) {
        throw new Error('RTAB payload offset must immediately follow the complete index');
    }
    if (payloadOffsetBig >= BigInt(bytes.length)) {
        throw new Error('RTAB payload section is empty or outside the file');
    }
    if (indexOffsetBig + indexSizeBig > BigInt(bytes.length)) {
        throw new Error('RTAB index range exceeds file size');
    }

    const indexOffset = ensureSafeNumber(indexOffsetBig, 'Index offset');
    const indexSize = ensureSafeNumber(indexSizeBig, 'Index size');
    const payloadOffset = ensureSafeNumber(payloadOffsetBig, 'Payload offset');
    const index = bytes.subarray(indexOffset, indexOffset + indexSize);
    if (index.toString('ascii', 0, 4) !== 'INDX') {
        throw new Error('RTAB index has invalid magic');
    }
    const entryCount = index.readUInt32LE(4);
    if (entryCount === 0) {
        throw new Error('RTAB bundle cannot contain zero assets');
    }
    if (index.readUInt16LE(8) !== RECORD_SIZE) {
        throw new Error(`Unsupported RTAB record size ${index.readUInt16LE(8)}; expected ${RECORD_SIZE}`);
    }
    if (index.readUInt16LE(10) !== 0) {
        throw new Error('Unsupported RTAB index flags');
    }

    const stringTableOffsetBig = readUInt64(index, 12, 'String-table offset');
    const stringTableSizeBig = readUInt64(index, 20, 'String-table size');
    const expectedStringOffset = BigInt(INDEX_HEADER_SIZE) + BigInt(entryCount) * BigInt(RECORD_SIZE);
    if (stringTableOffsetBig !== expectedStringOffset) {
        throw new Error('RTAB string table must immediately follow the record table');
    }
    ensureUint32(stringTableSizeBig, 'String-table size');
    const expectedIndexSize = expectedStringOffset + stringTableSizeBig + BigInt(DIGEST_SIZE);
    if (expectedIndexSize !== indexSizeBig) {
        throw new Error('RTAB index size does not match its header, records, string table, and digest');
    }

function validateManifest(manifest, registry) {
    if (!manifest || manifest.manifestSchemaVersion !== '1.1'
        || !Array.isArray(manifest.assets) || !Array.isArray(manifest.bundles)) {
        throw new Error('RTAB asset manifest has an invalid shape or manifestSchemaVersion');
    }
    assertOnlyKeys(manifest, ['manifestSchemaVersion', 'assets', 'bundles'], 'RTAB asset manifest');

    const expectedBundleByAssetId = new Map();
    const expectedBundles = new Map();
    for (const asset of registry.assets) {
        const categoryName = asset.logicalPath.split('/')[1];
        const bundleId = normalizeCategoryName(categoryName);
        const bundleFile = `/assets/${bundleId}.bin`;
        expectedBundleByAssetId.set(asset.assetId, bundleId);
        expectedBundles.set(bundleId, {
            bundleId,
            bundleFile,
            assets: [...(expectedBundles.get(bundleId)?.assets ?? []), asset]
        });
    }
    if (manifest.assets.length !== registry.assets.length) {
        throw new Error('RTAB asset manifest does not contain exactly the registered assets');
    }

    const manifestAssetsById = new Map(manifest.assets.map((asset) => [asset.assetId, asset]));
    if (manifestAssetsById.size !== manifest.assets.length) {
        throw new Error('RTAB asset manifest contains duplicate asset IDs');
    }
    for (const expected of registry.assets) {
        const actual = manifestAssetsById.get(expected.assetId);
        assertOnlyKeys(
            actual,
            ['assetId', 'logicalPath', 'textureKey', 'aliases', 'contentType', 'sourceSha256', 'bundleId'],
            `RTAB manifest asset "${expected.assetId}"`
        );
        if (!actual || actual.logicalPath !== expected.logicalPath
            || actual.textureKey !== expected.textureKey
            || actual.contentType !== expected.contentType
            || actual.bundleId !== expectedBundleByAssetId.get(expected.assetId)
            || JSON.stringify(actual.aliases) !== JSON.stringify(expected.aliases)) {
            throw new Error(`RTAB manifest metadata does not match registry asset "${expected.assetId}"`);
        }
        assertHexDigest(actual.sourceSha256, `source for "${expected.assetId}"`);
    }

    if (manifest.bundles.length !== expectedBundles.size) {
        throw new Error('RTAB manifest does not contain exactly the non-empty source categories');
    }
    const manifestBundlesById = new Map(manifest.bundles.map((bundle) => [bundle.bundleId, bundle]));
    if (manifestBundlesById.size !== manifest.bundles.length) {
        throw new Error('RTAB asset manifest contains duplicate bundle IDs');
    }
    for (const [bundleId, expected] of expectedBundles) {
        const actual = manifest.bundles.find((bundle) => bundle.bundleId === bundleId);
        assertOnlyKeys(
            actual,
            ['bundleId', 'bundleFile', 'formatVersion', 'assetCount', 'bundleSha256'],
            `RTAB manifest bundle "${bundleId}"`
        );
        if (!actual || actual.bundleFile !== expected.bundleFile
            || actual.formatVersion !== '1.0'
            || actual.assetCount !== expected.assets.length) {
            throw new Error(`RTAB manifest bundle metadata does not match derived category "${bundleId}"`);
        }
        assertHexDigest(actual.bundleSha256, `bundle "${bundleId}"`);
    }
}

async function findStaleBundlePaths(projectRoot, nextManifest) {
    const publicRoot = path.join(projectRoot, 'public');
    const currentPaths = new Set(nextManifest.bundles.map((bundle) => bundle.bundleFile.toLowerCase()));
    const stale = new Set();
    const legacyNestedShips = path.join(publicRoot, 'assets', 'Ships', 'ships.bin');
    const oldManifestPath = path.join(publicRoot, 'assets', 'asset-manifest.json');
    let oldManifest;
    try {
        oldManifest = JSON.parse(await readFile(oldManifestPath, 'utf8'));
    } catch (error) {
        if (error.code !== 'ENOENT') {
            throw new Error(`Could not inspect previous asset manifest for stale outputs: ${error.message}`);
        }
    }

    if (oldManifest !== undefined) {
        if (!oldManifest || !Array.isArray(oldManifest.bundles)) {
            throw new Error('Previous asset manifest does not declare a bundle list; refusing stale-output cleanup');
        }
        for (const bundle of oldManifest.bundles) {
            if (typeof bundle.bundleFile !== 'string'
                || !bundle.bundleFile.startsWith('/assets/')
                || bundle.bundleFile.includes('\\')
                || !bundle.bundleFile.toLowerCase().endsWith('.bin')
                || bundle.bundleFile.split('/').slice(1).some((part) => !part || part === '.' || part === '..')) {
                throw new Error('Previous asset manifest contains an unsafe bundle path; refusing stale-output cleanup');
            }
            if (!currentPaths.has(bundle.bundleFile.toLowerCase())) {
                const destination = path.resolve(publicRoot, `.${bundle.bundleFile}`);
                if (!destination.startsWith(`${publicRoot}${path.sep}`)) {
                    throw new Error(`Previous bundle path escapes public output root: ${bundle.bundleFile}`);
                }
                if (typeof bundle.bundleSha256 !== 'string' || !/^[a-f0-9]{64}$/.test(bundle.bundleSha256)) {
                    throw new Error(`Previous bundle "${bundle.bundleFile}" has no valid SHA-256; refusing stale-output cleanup`);
                }
                const info = await lstat(destination).catch((error) => {
                    if (error.code === 'ENOENT') {
                        return null;
                    }
                    throw error;
                });
                if (info) {
                    if (info.isSymbolicLink() || !info.isFile()) {
                        throw new Error(`Previous bundle is not a regular file: ${destination}`);
                    }
                    const bytes = await readFile(destination);
                    if (createHash('sha256').update(bytes).digest('hex') !== bundle.bundleSha256) {
                        throw new Error(`Previous bundle hash does not match its manifest; refusing stale-output cleanup: ${destination}`);
                    }
                }
                stale.add(destination);
            }

        const legacyInfo = await lstat(legacyNestedShips).catch((error) => {

        if (legacyInfo) {

        }
    }
    return [...stale];

function crc32(bytes: Buffer) {
    let crc = 0xffffffff;
    for (const byte of bytes) {
        crc ^= byte;
        for (let bit = 0; bit < 8; bit += 1) {
            crc = (crc & 1) ? (0xedb88320 ^ (crc >>> 1)) : (crc >>> 1);
        }
    }
    return (crc ^ 0xffffffff) >>> 0;
}

function pngChunk(type: string, data: Buffer) {
    const typeBytes = Buffer.from(type, 'ascii');
    const chunk = Buffer.alloc(12 + data.length);
    chunk.writeUInt32BE(data.length, 0);
    typeBytes.copy(chunk, 4);
    data.copy(chunk, 8);
    chunk.writeUInt32BE(crc32(chunk.subarray(4, 8 + data.length)), 8 + data.length);
    return chunk;
}

function createPng({
    width = 1,
    height = 1,
    bitDepth = 8,
    colorType = 6,
    interlace = 0,
    filter = 0
}: {
    width?: number;
    height?: number;
    bitDepth?: number;
    colorType?: number;
    interlace?: number;
    filter?: number;
} = {}) {
    const header = Buffer.alloc(13);
    header.writeUInt32BE(width, 0);
    header.writeUInt32BE(height, 4);
    header[8] = bitDepth;
    header[9] = colorType;
    header[10] = 0;
    header[11] = 0;
    header[12] = interlace;
    const scanlines = Buffer.alloc((width * 4 + 1) * height);
    for (let row = 0; row < height; row += 1) {
        scanlines[row * (width * 4 + 1)] = filter;
    }
    return Buffer.concat([
        Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]),
        pngChunk('IHDR', header),
        pngChunk('IDAT', deflateSync(scanlines)),
        pngChunk('IEND', Buffer.alloc(0))
    ]);
}

function hash(bytes: Buffer) {
    return createHash('sha256').update(bytes).digest('hex');
}

function changeFirstRecordContentType(bytes: Buffer, code: number) {
    const indexSize = Number(bytes.readBigUInt64LE(24));
    const index = bytes.subarray(64, 64 + indexSize);
    index.writeUInt16LE(code, 28 + 32);
    const digestOffset = index.length - 32;
    createHash('sha256').update(index.subarray(0, digestOffset)).digest().copy(index, digestOffset);
}

async function fileState(filePath) {
    let info;
    try {
        info = await lstat(filePath);
    } catch (error) {
        if (error.code === 'ENOENT') {
            return { exists: false, sha256: null };
        }
        throw error;
    }
    if (info.isSymbolicLink() || !info.isFile()) {
        throw new Error(`Asset publication path is not a regular file: ${filePath}`);

async function getProcessStartIdentity() {
    const estimatedStart = Date.now() - Math.round(process.uptime() * 1000);

async function acquireLock(projectRoot) {

        throw new Error(`Private source root exists but is not a real directory: ${privateRoot}`);

    const lockPath = path.join(privateRoot, LOCK_NAME);
    const owner = {
        pid: process.pid,
        hostname: hostname(),
        token: randomUUID(),
        startedAt: new Date().toISOString(),
        processStartIdentity: await getProcessStartIdentity()
    };

            throw error;

async function scanTransactionArtifacts(directory, artifacts = []) {

            return artifacts;
        }
        throw error;
    }
    entries.sort((left, right) => left.name < right.name ? -1 : left.name > right.name ? 1 : 0);
    for (const entry of entries) {
        const entryPath = path.join(directory, entry.name);
        const info = await lstat(entryPath);

async function readJournal(projectRoot) {
    const journalPath = path.join(projectRoot, 'private', JOURNAL_NAME);
    let bytes;
    try {
        bytes = await readFile(journalPath);

function validateJournalEntry(projectRoot, entry, transactionId, stale = false) {
    if (!entry || typeof entry.destination !== 'string'
        || typeof entry.priorExists !== 'boolean'
        || (entry.priorExists && !/^[a-f0-9]{64}$/.test(entry.priorSha256))
        || (!entry.priorExists && entry.priorSha256 !== null)
        || !/^[a-f0-9]{64}$/.test(entry.newSha256 ?? entry.priorSha256 ?? '')) {
        if (!(stale && entry && typeof entry.destination === 'string'
            && typeof entry.priorExists === 'boolean'
            && (entry.priorExists ? /^[a-f0-9]{64}$/.test(entry.priorSha256) : entry.priorSha256 === null))) {
            throw new Error('Asset publication journal contains an invalid output entry');
        }
    }
    const destination = resolveJournalPath(projectRoot, entry.destination, 'destination');
    const isPublic = destination.startsWith(`${path.join(projectRoot, 'public', 'assets')}${path.sep}`);
    const isGenerated = destination.startsWith(`${path.join(projectRoot, 'src', 'generated')}${path.sep}`);
    if (!isPublic && (!isGenerated || stale)) {
        throw new Error(`Asset publication journal destination is outside generated output roots: ${destination}`);
    }
    let temporary = null;
    let backup = null;
    if (!stale) {
        if (entry.unchanged) {
            if (entry.temporary !== null || entry.backup !== null
                || !entry.priorExists || entry.priorSha256 !== entry.newSha256) {
                throw new Error(`Asset publication journal has inconsistent unchanged output metadata for ${destination}`);
            }
            return { ...entry, destination, temporary: null, backup: null };
        }
        temporary = resolveJournalPath(projectRoot, entry.temporary, 'temporary');
        backup = entry.backup === null
            ? null
            : resolveJournalPath(projectRoot, entry.backup, 'backup');
        const expectedPrefix = `${destination}.rtab-${transactionId}.`;
        if (!temporary.startsWith(expectedPrefix) || !temporary.endsWith('.tmp')
            || (entry.priorExists
                ? (!backup || !backup.startsWith(expectedPrefix) || !backup.endsWith('.bak'))
                : backup !== null)
            || !/^[a-f0-9]{64}$/.test(entry.newSha256)) {
            throw new Error(`Asset publication journal has unsafe temporary or backup paths for ${destination}`);
        }
    }
    return { ...entry, destination, temporary, backup };
}

});

function isAllowedStateTransition(from, to) {
    return STATE_TRANSITIONS[from]?.includes(to) ?? false;
}

function isAllowedOperation(operation, state) {
    return OPERATION_STATES[operation]?.includes(state) ?? false;
}

async function appendJournal(journal, fields) {
    const payload = {
        ...fields,
        transactionId: journal.transactionId,
        sequence: journal.nextSequence
    };
    const record = signedJournalRecord(payload);
    const handle = await open(journal.journalPath, 'a');
    try {
        await handle.writeFile(`${JSON.stringify(record)}\n`);
        await handle.sync();
    } finally {
        await handle.close();
    }
    journal.records.push(record);
    journal.nextSequence += 1;
    if (record.type === 'state') {
        journal.state = record.to;
    }
    return record;
}

async function appendJournalState(journal, to) {
    if (journal.state === to) {
        return;
    }
    if (!isAllowedStateTransition(journal.state, to)) {
        throw new Error(`Invalid asset transaction state transition ${journal.state} -> ${to}`);
    }
    await appendJournal(journal, {
        type: 'state',
        from: journal.state,
        to,
        ...(to === 'committed' ? { manifestCommitted: true } : {})
    });
}

async function runJournaledOperation(journal, operation, target, expectedBefore, after, mutate) {
    if (!isAllowedOperation(operation, journal.state)) {
        throw new Error(`Asset operation "${operation}" is not allowed in transaction state "${journal.state}"`);
    }
    const before = await fileState(target);
    if (!sameFileState(before, expectedBefore)) {
        throw new Error(`Asset operation "${operation}" found an unexpected before-state; preserving it: ${target}`);
    }
    const operationId = randomUUID();
    const intent = {
        type: 'operation-intent',
        operationId,
        operation,
        target: relativePath(path.dirname(path.dirname(journal.journalPath)), target),
        before: expectedBefore,
        after
    };
    await appendJournal(journal, intent);
    await mutate();
    const actual = await fileState(target);
    if (!sameFileState(actual, after)) {
        throw new Error(`Asset operation "${operation}" did not produce its journaled after-state: ${target}`);
    }
    await appendJournal(journal, {
        type: 'operation-complete',
        operationId,
        operation,
        target: intent.target,
        result: 'applied'
    });
}

async function removeExact(filePath, expectedHash, label) {
    const state = await fileState(filePath);
    if (!state.exists) {
        return;
    }
    if (expectedHash && state.sha256 !== expectedHash) {
        throw new Error(`${label} has unexpected contents; preserving it: ${filePath}`);
    }
    await unlink(filePath);
}

async function createPublicationLink(source, destination, linkFile) {
    try {
        await linkFile(source, destination);
    } catch (error) {
        if (['EXDEV', 'EPERM', 'ENOTSUP', 'EOPNOTSUPP'].includes(error.code)) {
            throw new Error(
                `Asset publication requires hard-link support on the same filesystem (${error.code}): ${destination}`,
                { cause: error }
            );
        }
        throw error;
    }
}

async function rollbackTransaction(projectRoot, journal, { preserveRecoveryArtifacts = false } = {}) {
    const { begin } = journal;
    const outputs = begin.outputs.map((entry) => validateJournalEntry(projectRoot, entry, begin.transactionId));
    const stale = begin.staleOutputs.map((entry) => validateJournalEntry(projectRoot, entry, begin.transactionId, true));
    const states = [];

    for (const entry of outputs) {
        if (entry.unchanged) {
            const unchangedState = await fileState(entry.destination);
            const temporaryState = entry.temporary
                ? await fileState(entry.temporary)
                : { exists: false, sha256: null };
            const backupState = entry.backup
                ? await fileState(entry.backup)
                : { exists: false, sha256: null };
            if (!unchangedState.exists || unchangedState.sha256 !== entry.newSha256
                || temporaryState.exists || backupState.exists) {
                throw new Error(`Unchanged asset output no longer matches the transaction: ${entry.destination}`);
            }
            continue;
        }
        const destinationState = await fileState(entry.destination);
        const backupState = entry.backup ? await fileState(entry.backup) : { exists: false, sha256: null };
        const temporaryState = await fileState(entry.temporary);

        if (backupState.exists && (!entry.priorExists || backupState.sha256 !== entry.priorSha256)) {
            throw new Error(`Asset output backup has unexpected contents; preserving it: ${entry.backup}`);
        }
        if (temporaryState.exists && temporaryState.sha256 !== entry.newSha256) {
            throw new Error(`Asset output temporary file is incomplete or unexpected; preserving it: ${entry.temporary}`);
        }
        const destinationIsPrior = destinationState.exists && entry.priorExists
            && destinationState.sha256 === entry.priorSha256;
        const destinationIsNew = destinationState.exists && destinationState.sha256 === entry.newSha256;
        if (destinationState.exists && !destinationIsPrior && !destinationIsNew) {
            throw new Error(`Asset output has unexpected contents during rollback; preserving it: ${entry.destination}`);
        }
        if (!destinationState.exists && entry.priorExists && !backupState.exists) {
            throw new Error(`Prior asset output and verified backup are both missing: ${entry.destination}`);
        }
        if (destinationIsNew && entry.priorExists && !backupState.exists) {
            throw new Error(`Cannot restore prior asset output because its verified backup is missing: ${entry.destination}`);
        }
        states.push({ entry, destinationState, backupState, temporaryState, destinationIsPrior });
    }

    for (const entry of stale) {
        const state = await fileState(entry.destination);
        if (state.exists !== entry.priorExists || (state.exists && state.sha256 !== entry.priorSha256)) {
            throw new Error(`Pending stale asset output changed during rollback; preserving it: ${entry.destination}`);
        }
    }

    if (journal.state !== 'rolling-back') {
        await appendJournalState(journal, 'rolling-back');
    }

    for (const state of [...states].reverse()) {
        const { entry, destinationState, backupState, temporaryState, destinationIsPrior } = state;
        if (entry.priorExists && !destinationIsPrior) {
            if (destinationState.exists) {
                await runJournaledOperation(
                    journal,
                    'remove-new-output',
                    entry.destination,
                    destinationState,
                    { exists: false, sha256: null },
                    () => unlink(entry.destination)
                );
            }
            await runJournaledOperation(
                journal,
                'restore-prior',
                entry.destination,
                { exists: false, sha256: null },
                { exists: true, sha256: entry.priorSha256 },
                () => createPublicationLink(entry.backup, entry.destination, link)
            );
        } else if (!entry.priorExists && destinationState.exists) {
            await runJournaledOperation(
                journal,
                'remove-new-output',
                entry.destination,
                destinationState,
                { exists: false, sha256: null },
                () => unlink(entry.destination)
            );
        }
        const restoredState = await fileState(entry.destination);
        if (entry.priorExists && (!restoredState.exists || restoredState.sha256 !== entry.priorSha256)) {
            throw new Error(`Prior asset output could not be restored exactly: ${entry.destination}`);
        }
        if (!entry.priorExists && restoredState.exists) {
            throw new Error(`New asset output remains after rollback: ${entry.destination}`);
        }
        if (backupState.exists && !preserveRecoveryArtifacts) {
            await runJournaledOperation(
                journal,
                'remove-backup',
                entry.backup,
                backupState,
                { exists: false, sha256: null },
                () => removeExact(entry.backup, entry.priorSha256, 'Asset output backup')
            );
        }
        if (temporaryState.exists && !preserveRecoveryArtifacts) {
            await runJournaledOperation(
                journal,
                'remove-temp',
                entry.temporary,
                temporaryState,
                { exists: false, sha256: null },
                () => removeExact(entry.temporary, entry.newSha256, 'Asset output temporary file')
            );
        }
    }
    if (!preserveRecoveryArtifacts) {
        await appendJournalState(journal, 'rolled-back');
        await unlink(journal.journalPath);
    }
}

async function validateCommittedTransaction(projectRoot, journal) {
    const { begin } = journal;
    const outputs = begin.outputs.map((entry) => validateJournalEntry(projectRoot, entry, begin.transactionId));
    const manifest = outputs.find((entry) => (
        entry.destination === path.resolve(projectRoot, ...MANIFEST_RELATIVE.split('/'))
    ));
    if (!manifest) {
        throw new Error('Asset publication journal does not include the new manifest');
    }
    const manifestState = await fileState(manifest.destination);
    if (!manifestState.exists || manifestState.sha256 !== manifest.newSha256) {
        return false;
    }
    for (const entry of outputs) {
        const state = await fileState(entry.destination);
        if (!state.exists || state.sha256 !== entry.newSha256) {
            if (journal.state !== 'committed' && journal.state !== 'retiring' && journal.state !== 'complete') {
                return false;
            }
            throw new Error(`Committed asset output does not match its journal hash: ${entry.destination}`);
        }
    }
    await validateCommittedAssetOutputs({ projectRoot });
    return true;
}

async function completeCommittedTransaction(projectRoot, journal) {
    if (!await validateCommittedTransaction(projectRoot, journal)) {
        throw new Error('Committed asset outputs are no longer valid; preserving transaction artifacts');
    }
    const { begin } = journal;
    if (journal.state === 'committed') {
        await appendJournalState(journal, 'retiring');
    }
    const outputs = begin.outputs.map((entry) => validateJournalEntry(projectRoot, entry, begin.transactionId));
    const stale = begin.staleOutputs.map((entry) => validateJournalEntry(projectRoot, entry, begin.transactionId, true));
    const staleStates = [];
    for (const entry of stale) {
        const state = await fileState(entry.destination);
        if (state.exists && (!entry.priorExists || state.sha256 !== entry.priorSha256)) {
            throw new Error(`Pending stale output is ambiguous; preserving transaction artifacts: ${entry.destination}`);
        }
        staleStates.push({ entry, state });
    }
    for (const entry of outputs) {
        if (entry.backup) {
            const state = await fileState(entry.backup);
            if (state.exists && (!entry.priorExists || state.sha256 !== entry.priorSha256)) {
                throw new Error(`Asset output backup is ambiguous; preserving it: ${entry.backup}`);
            }
        }
        if (entry.temporary) {
            const state = await fileState(entry.temporary);
            if (state.exists && state.sha256 !== entry.newSha256) {
                throw new Error(`Asset output temporary file is ambiguous; preserving it: ${entry.temporary}`);
            }
        }
    }
    for (const { entry, state } of staleStates) {
        if (state.exists) {
            await runJournaledOperation(
                journal,
                'retire-stale',
                entry.destination,
                state,
                { exists: false, sha256: null },
                () => unlink(entry.destination)
            );
        }
    }
    for (const entry of outputs) {
        if (entry.backup) {
            const state = await fileState(entry.backup);
            if (state.exists) {
                await runJournaledOperation(
                    journal,
                    'remove-backup',
                    entry.backup,
                    state,
                    { exists: false, sha256: null },
                    () => removeExact(entry.backup, entry.priorExists ? entry.priorSha256 : null, 'Asset output backup')
                );
            }
        }
        if (entry.temporary) {
            const state = await fileState(entry.temporary);
            if (state.exists) {
                await runJournaledOperation(
                    journal,
                    'remove-temp',
                    entry.temporary,
                    state,
                    { exists: false, sha256: null },
                    () => removeExact(entry.temporary, entry.newSha256, 'Asset output temporary file')
                );
            }
        }
    }
    await appendJournalState(journal, 'complete');
    await unlink(journal.journalPath);
}

async function findTransactionArtifacts(projectRoot) {
    const roots = [
        path.join(projectRoot, 'private'),
        path.join(projectRoot, 'public', 'assets'),
        path.join(projectRoot, 'src', 'generated')
    ];
    const artifacts = [];
    for (const root of roots) {
        await scanTransactionArtifacts(root, artifacts);
    }
    return artifacts;
}

export async function assertNoPendingAssetOutputTransaction(projectRoot) {
    const privateRoot = path.join(projectRoot, 'private');
    const lockPath = path.join(privateRoot, LOCK_NAME);
    const journalPath = path.join(privateRoot, JOURNAL_NAME);
    for (const [filePath, label] of [[lockPath, 'lock'], [journalPath, 'journal']]) {
        try {
            const info = await lstat(filePath);
            if (info.isSymbolicLink() || !info.isFile()) {
                throw new Error(`Clean-clone validation found a non-regular asset publication ${label}: ${filePath}`);
            }
            throw new Error(`Clean-clone validation found an asset publication ${label}; preserving it without recovery: ${filePath}`);
        } catch (error) {
            if (error.code !== 'ENOENT') {
                throw error;
            }
        }
    }

    const artifacts = await findTransactionArtifacts(projectRoot);
    if (artifacts.length > 0) {
        throw new Error(`Clean-clone validation found an asset publication artifact; preserving it without recovery: ${artifacts[0]}`);
    }
}

export async function recoverAssetOutputTransaction(projectRoot) {
    const journal = await readJournal(projectRoot);
    const artifacts = await findTransactionArtifacts(projectRoot);
    if (!journal) {
        if (artifacts.length > 0) {
            throw new Error(`Orphan asset publication temporary or backup file found; inspect manually: ${artifacts[0]}`);
        }
        return { recovered: false };
    }

    const { begin } = journal;
    const outputs = begin.outputs.map((entry) => validateJournalEntry(projectRoot, entry, begin.transactionId));
    const destinations = new Set();
    const expectedArtifacts = new Set();
    const expectedOperationTargets = new Set();
    for (const entry of outputs) {
        if (destinations.has(entry.destination)) {
            throw new Error(`Asset publication journal repeats an output destination: ${entry.destination}`);
        }
        destinations.add(entry.destination);
        expectedOperationTargets.add(entry.destination);
        for (const artifact of [entry.temporary, entry.backup]) {
            if (artifact) {
                expectedOperationTargets.add(artifact);
                if (expectedArtifacts.has(artifact)) {
                    throw new Error(`Asset publication journal repeats a temporary or backup path: ${artifact}`);
                }
                expectedArtifacts.add(artifact);
            }
        }
    }
    const staleDestinations = new Set();
    for (const entry of begin.staleOutputs.map((item) => validateJournalEntry(projectRoot, item, begin.transactionId, true))) {
        if (destinations.has(entry.destination) || staleDestinations.has(entry.destination)) {
            throw new Error(`Asset publication journal has conflicting stale-output ownership: ${entry.destination}`);
        }
        staleDestinations.add(entry.destination);
        expectedOperationTargets.add(entry.destination);
    }
    for (const record of journal.records) {
        if (record.type === 'operation-intent') {
            const target = resolveJournalPath(projectRoot, record.target, 'operation target');
            if (!expectedOperationTargets.has(target)) {
                throw new Error(`Asset publication journal operation targets an unowned path; preserving it: ${target}`);
            }
        }
    }
    for (const artifact of artifacts) {
        if (!expectedArtifacts.has(artifact)) {
            throw new Error(`Unreferenced asset publication artifact is ambiguous; preserving it: ${artifact}`);
        }
    }

    if (journal.pendingOperation) {
        const operation = journal.pendingOperation;
        const target = resolveJournalPath(projectRoot, operation.target, 'operation target');
        const actual = await fileState(target);
        let result;
        if (sameFileState(actual, operation.after)) {
            result = 'applied';
        } else if (sameFileState(actual, operation.before)) {
            result = 'not-applied';
        } else {
            throw new Error(`Interrupted asset operation has an unexpected filesystem state; preserving journal and artifacts: ${target}`);
        }
        await appendJournal(journal, {
            type: 'operation-complete',
            operationId: operation.operationId,
            operation: operation.operation,
            target: operation.target,
            result
        });
        journal.pendingOperation = null;
    }

    const committedOutputsValid = await validateCommittedTransaction(projectRoot, journal);
    if (['committed', 'retiring', 'complete'].includes(journal.state) && !committedOutputsValid) {
        throw new Error('Committed asset transaction outputs are invalid; preserving journal, backups, and uncertain artifacts');
    }
    if (committedOutputsValid && ['preparing', 'staging', 'publishing', 'validating'].includes(journal.state)) {
        while (journal.state !== 'validating') {
            const nextState = {
                preparing: 'staging',
                staging: 'publishing',
                publishing: 'validating'
            }[journal.state];
            if (!nextState) {
                throw new Error(`Cannot validate recovered transaction from state "${journal.state}"`);
            }
            await appendJournalState(journal, nextState);
        }
        await appendJournalState(journal, 'committed');
    }
    if (committedOutputsValid && ['committed', 'retiring', 'complete'].includes(journal.state)) {
        await completeCommittedTransaction(projectRoot, journal);
        return { recovered: true, mode: 'completed-commit' };
    }
    await rollbackTransaction(projectRoot, journal);
    return { recovered: true, mode: 'rolled-back' };
}

export async function withAssetOutputLock(projectRoot, action) {
    const lock = await acquireLock(projectRoot);
    try {
        await recoverAssetOutputTransaction(projectRoot);
        return await action();
    } finally {
        await releaseLock(lock);
    }
}

function toOutputBytes(value) {
    return Buffer.isBuffer(value) ? value : Buffer.from(value, 'utf8');
}

function outputOrder([leftPath], [rightPath]) {
    const leftManifest = path.basename(leftPath) === 'asset-manifest.json';
    const rightManifest = path.basename(rightPath) === 'asset-manifest.json';
    if (leftManifest !== rightManifest) {
        return leftManifest ? 1 : -1;
    }
    return leftPath < rightPath ? -1 : leftPath > rightPath ? 1 : 0;
}

async function stageOutput(entry, bytes) {
    await writeFile(entry.temporary, bytes, { flag: 'wx' });
    await syncFile(entry.temporary);
    const state = await fileState(entry.temporary);
    if (!state.exists || state.sha256 !== entry.newSha256) {
        throw new Error(`Staged asset output failed SHA-256 verification: ${entry.temporary}`);
    }
}

export async function publishAssetOutputs({
    projectRoot,
    outputs,
    stalePaths = [],
    linkFile = link,
    validatePublishedOutputs = async () => {}
}) {
    const transactionId = randomUUID();
    const outputEntries = [];
    const staleEntries = [];
    for (const [destination, contents] of [...outputs].sort(outputOrder)) {
        const absoluteDestination = path.resolve(destination);
        await assertSafeDestination(projectRoot, absoluteDestination, true);
        const newBytes = toOutputBytes(contents);
        const prior = await fileState(absoluteDestination);
        const unchanged = prior.exists && prior.sha256 === hash(newBytes);
        const prefix = `${absoluteDestination}.rtab-${transactionId}`;
        outputEntries.push({
            destination: absoluteDestination,
            priorExists: prior.exists,
            priorSha256: prior.sha256,
            temporary: unchanged ? null : `${prefix}.tmp`,
            backup: unchanged || !prior.exists ? null : `${prefix}.bak`,
            newSha256: hash(newBytes),
            bytes: newBytes,
            unchanged
        });
    }

    const outputPaths = new Set(outputEntries.map((entry) => entry.destination));
    for (const stalePath of stalePaths) {
        const destination = path.resolve(stalePath);
        await assertSafeDestination(projectRoot, destination, false);
        if (outputPaths.has(destination)) {
            continue;
        }
        const prior = await fileState(destination);
        staleEntries.push({
            destination: relativePath(projectRoot, destination),
            priorExists: prior.exists,
            priorSha256: prior.sha256
        });
    }

    const journalPath = path.join(projectRoot, 'private', JOURNAL_NAME);
    const begin = signedJournalRecord({
        type: 'begin',
        transactionId,
        sequence: 0,
        state: 'preparing',
        outputs: outputEntries.map((entry) => ({
            destination: relativePath(projectRoot, entry.destination),
            priorExists: entry.priorExists,
            priorSha256: entry.priorSha256,
            temporary: entry.temporary ? relativePath(projectRoot, entry.temporary) : null,
            backup: entry.backup ? relativePath(projectRoot, entry.backup) : null,
            newSha256: entry.newSha256,
            unchanged: entry.unchanged
        })),
        staleOutputs: staleEntries
    });
    const handle = await open(journalPath, 'wx');
    try {
        await handle.writeFile(`${JSON.stringify(begin)}\n`);
        await handle.sync();
    } finally {
        await handle.close();
    }
    const journal = {
        journalPath,
        transactionId,
        begin,
        records: [begin],
        state: 'preparing',
        pendingOperation: null,
        nextSequence: 1
    };

    let postPublicationValidationFailed = false;
    try {
        await appendJournalState(journal, 'staging');
        for (const entry of outputEntries) {
            if (entry.unchanged) {
                continue;
            }
            await runJournaledOperation(
                journal,
                'stage-temp',
                entry.temporary,
                { exists: false, sha256: null },
                { exists: true, sha256: entry.newSha256 },
                () => stageOutput(entry, entry.bytes)
            );
        }

        await appendJournalState(journal, 'publishing');
        for (const entry of outputEntries) {
            if (entry.unchanged) {
                continue;
            }
            const prior = await fileState(entry.destination);
            if (entry.priorExists) {
                if (!prior.exists || prior.sha256 !== entry.priorSha256) {
                    throw new Error(`Asset output changed after publication preflight: ${entry.destination}`);
                }
                await runJournaledOperation(
                    journal,
                    'backup-prior',
                    entry.backup,
                    { exists: false, sha256: null },
                    { exists: true, sha256: entry.priorSha256 },
                    () => createPublicationLink(entry.destination, entry.backup, linkFile)
                );
                const stillPrior = await fileState(entry.destination);
                if (!stillPrior.exists || stillPrior.sha256 !== entry.priorSha256) {
                    throw new Error(`Asset output changed while its backup was being created: ${entry.destination}`);
                }
                await runJournaledOperation(
                    journal,
                    'remove-destination',
                    entry.destination,
                    { exists: true, sha256: entry.priorSha256 },
                    { exists: false, sha256: null },
                    () => unlink(entry.destination)
                );
            } else if (prior.exists) {
                throw new Error(`Asset output appeared after publication preflight: ${entry.destination}`);
            }
            await runJournaledOperation(
                journal,
                'publish-output',
                entry.destination,
                { exists: false, sha256: null },
                { exists: true, sha256: entry.newSha256 },
                () => createPublicationLink(entry.temporary, entry.destination, linkFile)
            );
        }

        await appendJournalState(journal, 'validating');
        try {
            await validatePublishedOutputs();
        } catch (error) {
            postPublicationValidationFailed = true;
            throw error;
        }
        if (!await validateCommittedTransaction(projectRoot, journal)) {
            throw new Error('New asset manifest was not committed as the validated output set');
        }
        await appendJournalState(journal, 'committed');
        await appendJournalState(journal, 'retiring');
        for (const stale of staleEntries) {
            const destination = path.resolve(projectRoot, ...stale.destination.split('/'));
            const current = await fileState(destination);
            if (current.exists !== stale.priorExists
                || (current.exists && current.sha256 !== stale.priorSha256)) {
                throw new Error(`Pending stale output changed before retirement: ${destination}`);
            }
            if (current.exists) {
                await runJournaledOperation(
                    journal,
                    'retire-stale',
                    destination,
                    current,
                    { exists: false, sha256: null },
                    () => unlink(destination)
                );
            }
        }
        for (const entry of outputEntries) {
            if (entry.backup) {
                const backupState = await fileState(entry.backup);
                if (backupState.exists) {
                    await runJournaledOperation(
                        journal,
                        'remove-backup',
                        entry.backup,
                        backupState,
                        { exists: false, sha256: null },
                        () => removeExact(entry.backup, entry.priorSha256, 'Asset output backup')
                    );
                }
            }
            if (entry.temporary) {
                const temporaryState = await fileState(entry.temporary);
                if (temporaryState.exists) {
                    await runJournaledOperation(
                        journal,
                        'remove-temp',
                        entry.temporary,
                        temporaryState,
                        { exists: false, sha256: null },
                        () => removeExact(entry.temporary, entry.newSha256, 'Asset output temporary file')
                    );
                }
            }
        }
        await appendJournalState(journal, 'complete');
        await unlink(journalPath);
        return { transactionId, outputFiles: outputEntries.length, staleFiles: staleEntries.length };
    } catch (error) {
        const pendingJournal = await readJournal(projectRoot);
        if (pendingJournal) {
            try {
                if (['committed', 'retiring', 'complete'].includes(pendingJournal.state)) {
                    throw new Error(
                        `Asset outputs are committed; preserving the journal for locked completion recovery: ${error.message}`,
                        { cause: error }
                    );
                }
                if (pendingJournal.pendingOperation) {
                    const operation = pendingJournal.pendingOperation;
                    const target = resolveJournalPath(projectRoot, operation.target, 'operation target');
                    const actual = await fileState(target);
                    let result;
                    if (sameFileState(actual, operation.after)) {
                        result = 'applied';
                    } else if (sameFileState(actual, operation.before)) {
                        result = 'not-applied';
                    } else {
                        throw new Error(`Interrupted asset operation has unexpected contents: ${target}`);
                    }
                    await appendJournal(pendingJournal, {
                        type: 'operation-complete',
                        operationId: operation.operationId,
                        operation: operation.operation,
                        target: operation.target,
                        result
                    });
                    pendingJournal.pendingOperation = null;
                }
                await rollbackTransaction(projectRoot, pendingJournal, {
                    preserveRecoveryArtifacts: postPublicationValidationFailed
                });
            } catch (rollbackError) {
                throw new AggregateError(
                    [error, rollbackError],
                    `Asset output publication failed and rollback was incomplete: ${rollbackError.message}`
                );
            }
        }
        if (postPublicationValidationFailed) {
            throw new Error(

const PNG_SIGNATURE = Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]);
const CRC_TABLE = new Uint32Array(256);

for (let index = 0; index < CRC_TABLE.length; index += 1) {
    let crc = index;
    for (let bit = 0; bit < 8; bit += 1) {
        crc = (crc & 1) ? (0xedb88320 ^ (crc >>> 1)) : (crc >>> 1);
    }
    CRC_TABLE[index] = crc >>> 0;
}

function crc32(bytes) {
    let crc = 0xffffffff;
    for (const byte of bytes) {
        crc = CRC_TABLE[(crc ^ byte) & 0xff] ^ (crc >>> 8);
    }
    return (crc ^ 0xffffffff) >>> 0;
}

function fail(asset, message) {
    throw new Error(`Fatal: PNG asset ${asset.logicalPath} ${message}`);
}

async function inflatePngChunks(chunks, expectedOutputBytes, compressedBytes, asset) {
    const inflater = createInflate();
    Readable.from(chunks).pipe(inflater);
    const outputChunks = [];
    let outputBytes = 0;
    try {
        for await (const output of inflater) {
            outputBytes += output.length;
            if (outputBytes > expectedOutputBytes) {
                inflater.destroy();
                fail(asset, `inflated scanline data exceeds its ${expectedOutputBytes}-byte limit`);
            }
            outputChunks.push(output);
        }
    } catch (error) {
        if (error.message.startsWith(`Fatal: PNG asset ${asset.logicalPath}`)) {
            throw error;
        }
        fail(asset, `could not be safely decompressed: ${error.message}`);
    }

    if (!inflater.readableEnded) {
        fail(asset, 'compressed IDAT stream did not terminate cleanly');
    }
    if (inflater.bytesWritten !== compressedBytes) {
        fail(asset, 'contains unused bytes after the compressed IDAT stream');
    }
    return Buffer.concat(outputChunks, outputBytes);
}

export async function validatePng(bytes, asset) {
    if (!Buffer.isBuffer(bytes) || bytes.length < PNG_SIGNATURE.length + 25) {
        fail(asset, 'is shorter than the minimum PNG structure');
    }
    if (bytes.length > PNG_LIMITS.maxFileBytes) {
        fail(asset, `exceeds the ${PNG_LIMITS.maxFileBytes}-byte compressed file limit`);
    }
    if (!bytes.subarray(0, PNG_SIGNATURE.length).equals(PNG_SIGNATURE)) {
        fail(asset, 'has an invalid PNG signature');
    }

    let offset = PNG_SIGNATURE.length;
    let width;
    let height;
    let sawHeader = false;
    let sawIdat = false;
    let idatEnded = false;
    let sawPalette = false;
    let sawEnd = false;
    const idatChunks = [];
    let compressedBytes = 0;

    while (offset < bytes.length) {
        if (offset + 12 > bytes.length) {
            fail(asset, 'contains a truncated PNG chunk');
        }
        const length = bytes.readUInt32BE(offset);
        const typeStart = offset + 4;
        const dataStart = offset + 8;
        const dataEnd = dataStart + length;
        const chunkEnd = dataEnd + 4;
        if (chunkEnd > bytes.length) {
            fail(asset, 'contains a PNG chunk outside the file bounds');
        }
        const type = bytes.toString('ascii', typeStart, typeStart + 4);
        if (!/^[A-Za-z]{4}$/.test(type) || (type.charCodeAt(2) & 0x20) !== 0) {
            fail(asset, `contains an invalid PNG chunk type "${type}"`);
        }
        const chunkBytes = bytes.subarray(typeStart, dataEnd);
        if (crc32(chunkBytes) !== bytes.readUInt32BE(dataEnd)) {
            fail(asset, `has an invalid CRC for PNG chunk ${type}`);
        }
        const data = bytes.subarray(dataStart, dataEnd);

        if (!sawHeader && type !== 'IHDR') {
            fail(asset, 'does not begin with IHDR');
        }
        if (type === 'IHDR') {
            if (sawHeader || length !== 13) {
                fail(asset, 'has a duplicate or malformed IHDR chunk');
            }
            sawHeader = true;
            width = data.readUInt32BE(0);
            height = data.readUInt32BE(4);
            const bitDepth = data[8];
            const colorType = data[9];
            const compressionMethod = data[10];
            const filterMethod = data[11];
            const interlaceMethod = data[12];
            if (width === 0 || height === 0
                || width > PNG_LIMITS.maxWidth || height > PNG_LIMITS.maxHeight) {
                fail(asset, `has dimensions outside the ${PNG_LIMITS.maxWidth}x${PNG_LIMITS.maxHeight} limit`);
            }
            if (bitDepth !== 8 || colorType !== 6) {
                fail(asset, `uses unsupported PNG profile (expected color type 6, bit depth 8; got ${colorType}/${bitDepth})`);
            }
            if (compressionMethod !== 0 || filterMethod !== 0 || interlaceMethod !== 0) {
                fail(asset, 'uses unsupported PNG compression, filter, or interlace profile (only method 0 and non-interlaced images are supported)');
            }
        } else if (type === 'IDAT') {
            if (idatEnded) {
                fail(asset, 'has non-contiguous IDAT chunks');
            }
            sawIdat = true;
            compressedBytes += length;
            if (compressedBytes > PNG_LIMITS.maxFileBytes) {
                fail(asset, `exceeds the ${PNG_LIMITS.maxFileBytes}-byte compressed IDAT limit`);
            }
            idatChunks.push(data);
        } else if (type === 'IEND') {
            if (length !== 0 || !sawIdat || sawEnd) {
                fail(asset, 'has a malformed or misplaced IEND chunk');
            }
            sawEnd = true;
        } else if (type === 'PLTE') {
            if (sawPalette || sawIdat || length < 3 || length > 768 || length % 3 !== 0) {
                fail(asset, 'has a malformed or misplaced PLTE chunk');
            }
            sawPalette = true;
        } else {
            if (sawIdat) {
                idatEnded = true;
            }
            const isCritical = (type.charCodeAt(0) & 0x20) === 0;
            if (isCritical && type !== 'PLTE') {
                fail(asset, `contains unsupported critical PNG chunk ${type}`);
            }
        }

        offset = chunkEnd;
        if (sawEnd) {
            if (offset !== bytes.length) {
                fail(asset, 'contains trailing bytes after IEND');
            }
            break;
        }
    }
    if (!sawHeader || !sawIdat || !sawEnd) {
        fail(asset, 'is missing IHDR, IDAT, or IEND');
    }

    const rowBytes = width * 4;
    const decodedPixelBytes = rowBytes * height;
    const scanlineBytes = (rowBytes + 1) * height;
    if (decodedPixelBytes > PNG_LIMITS.maxDecodedPixelBytes) {
        fail(asset, `exceeds the ${PNG_LIMITS.maxDecodedPixelBytes}-byte per-image decoded pixel limit`);
    }
    if (scanlineBytes > PNG_LIMITS.maxInflatedScanlineBytes) {
        fail(asset, `exceeds the ${PNG_LIMITS.maxInflatedScanlineBytes}-byte per-image inflated scanline limit`);

        }
    }

        width,
        height,
        compressedBytes,
        decodedPixelBytes,
        scanlineBytes
    };

const SIGNATURE = Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]);

function crc32(bytes: Buffer) {
    let crc = 0xffffffff;
    for (const byte of bytes) {
        crc ^= byte;
        for (let bit = 0; bit < 8; bit += 1) {
            crc = (crc & 1) ? (0xedb88320 ^ (crc >>> 1)) : (crc >>> 1);
        }
    }
    return (crc ^ 0xffffffff) >>> 0;
}

function chunk(type: string, data: Buffer) {
    const typeBytes = Buffer.from(type, 'ascii');
    const output = Buffer.alloc(12 + data.length);
    output.writeUInt32BE(data.length, 0);
    typeBytes.copy(output, 4);
    data.copy(output, 8);
    output.writeUInt32BE(crc32(output.subarray(4, 8 + data.length)), 8 + data.length);
    return output;
}

function header(width = 1, height = 1) {
    const data = Buffer.alloc(13);
    data.writeUInt32BE(width, 0);
    data.writeUInt32BE(height, 4);
    data[8] = 8;
    data[9] = 6;
    return chunk('IHDR', data);
}

function png({
    width = 1,
    height = 1,
    raw,
    idatParts,
    beforeIdat = [],
    afterIdat = [],
    trailing = Buffer.alloc(0)
}: {
    width?: number;
    height?: number;
    raw?: Buffer;
    idatParts?: Buffer[];
    beforeIdat?: Buffer[];
    afterIdat?: Buffer[];
    trailing?: Buffer;
} = {}) {
    const compressed = deflateSync(raw ?? Buffer.alloc((width * 4 + 1) * height));
    const parts = idatParts ?? [compressed];
    return Buffer.concat([
        SIGNATURE,
        header(width, height),
        ...beforeIdat,
        ...parts.map((part) => chunk('IDAT', part)),
        ...afterIdat,
        chunk('IEND', Buffer.alloc(0)),
        trailing
    ]);
}

function corruptChunkCrc(bytes: Buffer, targetType: string) {
    const output = Buffer.from(bytes);
    let offset = SIGNATURE.length;
    while (offset + 12 <= output.length) {
        const length = output.readUInt32BE(offset);
        const type = output.toString('ascii', offset + 4, offset + 8);
        if (type === targetType) {
            output[offset + 8 + length] ^= 0xff;
            return output;
        }
        offset += 12 + length;
    }
    throw new Error(`Missing test chunk ${targetType}`);
}
